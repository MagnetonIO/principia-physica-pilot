\begin{abstract}
We formulate compositional physical modelling as a typed relation between experimental interfaces and mathematical systems. Symmetric monoidal categories encode sequential and parallel composition; symmetric monoidal functors encode models. We prove the resulting substitution and congruence principles, distinguish structural equivalence from physical sameness, and exhibit counterexamples to the inference from mathematical admissibility to empirical realization. Every physical assertion is isolated as an empirical statement or an explicit assumption.
\end{abstract}

\section{Scope and epistemic discipline}
The thesis motivating this paper is that physics studies empirically realized compositional mathematical structures. The expression has two logically independent parts. Composition is a mathematical constraint. Realization is an empirical relation between a model and interventions, preparations, or observations. No categorical theorem can establish the latter without empirical premises.

\begin{definition}[Interface theory]
An \emph{interface theory} is a symmetric monoidal category $(\mathcal E,\otimes,I,\sigma)$. Objects are experimental interface types; morphisms $e:X\to Y$ are admissible experimental processes; composition represents sequential connection; and $\otimes$ represents parallel juxtaposition.
\end{definition}

\begin{definition}[Mathematical system theory]
A \emph{system theory} is a symmetric monoidal category $(\mathcal M,\boxtimes,\mathbf 1)$. Its objects are mathematical state or boundary types and its morphisms are mathematical processes.
\end{definition}

\begin{definition}[Compositional model]
A compositional model is a strong symmetric monoidal functor
\[
F:\mathcal E\longrightarrow\mathcal M,
\]
with coherent isomorphisms $F(X)\boxtimes F(Y)\cong F(X\otimes Y)$ and $\mathbf 1\cong F(I)$.
\end{definition}

These are definitions, not empirical claims. They specify a class of possible representations.

\begin{assumption}[Typed experimental composition]
For the domain under study, laboratory procedures admit repeatable interface types, and the connections represented by composition and tensor product are operationally implementable to stated tolerances.
\end{assumption}

\begin{empirical}[Realization input]
Whether a particular laboratory domain satisfies the preceding assumption, and whether a proposed $F$ predicts its observed statistics within uncertainty, must be decided from calibration and outcome data.
\end{empirical}

\section{Consequences of functorial modelling}
\begin{lemma}[Substitution]
Let $F:\mathcal E\to\mathcal M$ be a functor. If $e=e':X\to Y$, then for all composable $a:W\to X$ and $b:Y\to Z$,
\[
F(b\circ e\circ a)=F(b\circ e'\circ a).
\]
\end{lemma}
\begin{proof}
Equality in a category is a congruence for composition, so $b\circ e\circ a=b\circ e'\circ a$. Applying the function that $F$ assigns to the hom-set $\mathcal E(W,Z)$ gives equal images. Functoriality then rewrites these images as $F(b)\circ F(e)\circ F(a)$ and $F(b)\circ F(e')\circ F(a)$.
\end{proof}

\begin{theorem}[Preservation of generated process equations]
Let $F:\mathcal E\to\mathcal M$ be a strong symmetric monoidal functor. Every equation between morphism expressions generated from identities, composition, tensor product, symmetries, and a fixed collection of equations in $\mathcal E$ is mapped to an equation in $\mathcal M$.
\end{theorem}
\begin{proof}
Proceed by induction on the derivation of an equation. Reflexivity, symmetry, and transitivity are preserved because each hom-set map is a function. The composition-congruence step is preserved by functoriality. The tensor-congruence step is preserved after conjugating by the monoidal coherence isomorphisms. Equations for identities and composition are the functor axioms. Equations involving associators, unitors, and symmetries are preserved by the coherence diagrams for a strong symmetric monoidal functor. Thus every generating rule and every inference rule has an equal image.
\end{proof}

\begin{corollary}[Contextual invariance]
If two experimental processes are equated in $\mathcal E$, no compositional context constructed from sequential or parallel connection distinguishes their images under $F$.
\end{corollary}
\begin{proof}
A context is a morphism expression with one hole. Insert the two equal morphisms. The resulting equation is generated by composition and tensor congruence, so the theorem applies.
\end{proof}

\begin{interpretation}
The corollary is a theorem about the chosen formalization. Treating its premise $e=e'$ as a claim about physical indistinguishability requires experimental evidence that the selected contexts include the interventions relevant to the intended claim.
\end{interpretation}

\section{States, effects, and probabilities}
\begin{definition}
In a monoidal category, a \emph{state} of $X$ is a morphism $s:I\to X$, an \emph{effect} is $p:X\to I$, and the closed experiment $p\circ s:I\to I$ is a scalar.
\end{definition}

\begin{assumption}[Probabilistic interpretation]
There is a map $\nu:\mathcal M(\mathbf 1,\mathbf 1)\to[0,1]$ such that the scalars assigned to the mutually exclusive outcomes of each specified measurement sum to one.
\end{assumption}

\begin{proposition}[Compositional prediction]
Given a state $s:I\to X$, a process $e:X\to Y$, and an effect $p:Y\to I$, a compositional model assigns the scalar
\[
F(p\circ e\circ s)=F(p)\circ F(e)\circ F(s).
\]
Under the probabilistic-interpretation assumption, its predicted outcome probability is $\nu(F(p\circ e\circ s))$.
\end{proposition}
\begin{proof}
The scalar identity is the composition axiom for $F$, applied twice. The probability statement is the content of the stated assumption, not a consequence of category theory.
\end{proof}

\begin{example}[Finite stochastic systems]
Let objects be finite sets and morphisms column-stochastic matrices. Matrix multiplication is sequential composition and the Kronecker product is parallel composition. A probability vector is a state and a row vector selecting an event is an effect. Normalization supplies the probabilistic interpretation. The categorical equations reduce to associativity of matrix multiplication and its compatibility with Kronecker products.
\end{example}

\section{Limits of structural reasoning}
\begin{counterexample}[Admissibility is not realization]
Let $\mathcal E$ be the free symmetric monoidal category on one process $g:X\to X$. Both assignments $F(g)=\mathrm{id}_{\mathbb R}$ and $G(g)=-\mathrm{id}_{\mathbb R}$ extend to strong symmetric monoidal functors into finite-dimensional real vector spaces. Hence both are mathematically admissible. If an experiment prepares $1$ and measures the sign after one use of $g$, the two models make different predictions. Category-theoretic admissibility alone selects neither model.
\end{counterexample}

\begin{counterexample}[Agreement on a restricted context]
Two smooth functions $f,g:\mathbb R\to\mathbb R$ can agree on all calibrated inputs $x\in[-1,1]$ while differing outside that interval; take $f(x)=0$ and a smooth $g$ vanishing on $[-1,1]$ but nonzero for $x>1$. Agreement in the tested subcategory does not entail global physical equivalence.
\end{counterexample}

\begin{definition}[Empirical realization relative to a test family]
Fix experiments $T$, observed estimates $\widehat q_t$, uncertainties $\epsilon_t$, and model predictions $q_t(F)$. The model $F$ is \emph{$(T,\epsilon)$-realized} when
\[
|q_t(F)-\widehat q_t|\leq \epsilon_t\qquad(t\in T).
\]
\end{definition}

\begin{proposition}[Monotonicity of realization]
If $F$ is $(T,\epsilon)$-realized, $T'\subseteq T$, and $\epsilon'_t\geq\epsilon_t$ for $t\in T'$, then $F$ is $(T',\epsilon')$-realized.
\end{proposition}
\begin{proof}
For each $t\in T'$, realization gives $|q_t(F)-\widehat q_t|\leq\epsilon_t$. The assumed inequality yields $|q_t(F)-\widehat q_t|\leq\epsilon'_t$.
\end{proof}

\begin{conjecture}[Modular identifiability]
For experimentally important system classes, identifiability of components plus an empirically validated independence condition can be promoted to identifiability of their monoidal composite. The conjecture is false without restrictions on hidden shared causes and interface completeness.
\end{conjecture}

\section{Dependency ledger and conclusion}
The mathematical results depend on the definitions of category, symmetric monoidal category, and strong symmetric monoidal functor, together with ordinary equality logic. Their application depends additionally on typed experimental composition, a probabilistic interpretation when probabilities are used, calibration data, and a specified test family. The theorem establishes compositional consistency; it does not establish realization.
