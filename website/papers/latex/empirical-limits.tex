\begin{abstract}
Mathematical consistency constrains physical models but does not establish their realization. We formalize finite observational records, empirical equivalence, identifiability, and scope. We prove a general finite-data underdetermination theorem, derive monotonicity and quotient results for observational equivalence, and give counterexamples to extrapolation and unrestricted causal inference. The analysis identifies which conclusions are mathematical, which depend on methodological assumptions, and which remain empirical.
\end{abstract}

\section{Data and models}
\begin{definition}[Experimental design]
An experimental design is a tuple $D=(X,Y,T,\delta)$, where $X$ is a set of admissible interventions or inputs, $Y$ an outcome space, $T\subseteq X$ the tested inputs, and $\delta$ records the measurement resolution or uncertainty specification.
\end{definition}

\begin{definition}[Model and prediction]
A deterministic model is a function $m:X\to Y$. More generally, a statistical model assigns a probability measure $P_m(\cdot\mid x)$ on $Y$ to each $x\in X$.
\end{definition}

\begin{definition}[Empirical equivalence]
Two deterministic models satisfy $m\sim_T n$ when $m(x)=n(x)$ for all $x\in T$. At finite resolution, write $m\sim_{T,\delta}n$ when their predictions differ by no more than the specified tolerance at every tested input. For statistical models, equality may be replaced by a declared distance between probability measures.
\end{definition}

\begin{definition}[Identifiability]
A model class $\mathcal M$ is identifiable under an observation operator $O$ when $O(m)=O(n)$ implies $m=n$ for $m,n\in\mathcal M$. It is identifiable up to an equivalence relation $\approx$ when the same premise implies $m\approx n$.
\end{definition}

\begin{assumption}[Record reliability]
Recorded inputs, outcomes, timestamps, and uncertainty estimates refer to the intended operations and have passed the stated calibration and data-integrity checks.
\end{assumption}

\begin{empirical}[Status of the record]
Record reliability is an empirical input. Mathematics can propagate a stated uncertainty model but cannot certify that instruments, labels, or sampling procedures satisfy it.
\end{empirical}

\section{Logical structure of finite tests}
\begin{lemma}
For fixed $T$, empirical equivalence $\sim_T$ is an equivalence relation on the set of deterministic models $X\to Y$.
\end{lemma}
\begin{proof}
Reflexivity holds because $m(x)=m(x)$. Symmetry follows from symmetry of equality. If $m\sim_T n$ and $n\sim_T r$, then $m(x)=n(x)=r(x)$ for every $x\in T$, so $m\sim_T r$.
\end{proof}

\begin{proposition}[Restriction monotonicity]
If $T'\subseteq T$ and $m\sim_T n$, then $m\sim_{T'}n$.
\end{proposition}
\begin{proof}
Every $x\in T'$ belongs to $T$, where the assumed equality holds.
\end{proof}

\begin{proposition}[Quotient of testable content]
Let $R_T:\mathcal M\to Y^T$ be restriction, $R_T(m)=m|_T$. The quotient set $\mathcal M/{\sim_T}$ is in bijection with the image $R_T(\mathcal M)$.
\end{proposition}
\begin{proof}
Define $\Phi([m])=R_T(m)$. If $[m]=[n]$, then $m\sim_Tn$, hence their restrictions agree, so $\Phi$ is well defined. Equal images imply $m\sim_Tn$, proving injectivity. Every element of $R_T(\mathcal M)$ is the restriction of some $m$, proving surjectivity.
\end{proof}

\begin{interpretation}
A finite experiment directly distinguishes equivalence classes, not unrestricted global models. Selecting a representative requires additional structure such as smoothness, a parametric family, a simplicity preference, or prior information.
\end{interpretation}

\section{A general underdetermination theorem}
\begin{assumption}[Perturbation closure]
The deterministic model class $\mathcal M$ contains a model $m$ and, for infinitely many nonzero perturbations $h:X\to V$, also contains $m+h$, where $Y$ lies in a vector space $V$ and each $h$ vanishes on the tested set $T$.
\end{assumption}

\begin{theorem}[Finite-data underdetermination]
Under perturbation closure, the observational record on $T$ does not identify $m$ uniquely in $\mathcal M$. If the admissible perturbations are pairwise distinct, the empirical-equivalence class of $m$ is infinite.
\end{theorem}
\begin{proof}
For every admissible $h$ and every $x\in T$, $(m+h)(x)=m(x)+h(x)=m(x)$. Hence $m+h\sim_Tm$. Nonzero perturbations give models distinct from $m$, and pairwise distinct perturbations give pairwise distinct models.
\end{proof}

\begin{corollary}[Smooth finite samples]
Let $X$ be a positive-dimensional smooth manifold, $Y=\mathbb R$, and $T\subset X$ finite. The class $C^\infty(X,\mathbb R)$ is not identifiable by restriction to $T$.
\end{corollary}
\begin{proof}
Choose a coordinate ball disjoint from $T$ and a nonzero smooth bump function $h$ supported in that ball. Every scalar multiple $\lambda h$ vanishes on $T$. Apply the theorem to obtain infinitely many alternatives $m+\lambda h$.
\end{proof}

\begin{example}[Polynomial restriction restores identification]
A real polynomial of degree at most $d$ is uniquely determined by its values at $d+1$ distinct points. If two such polynomials agree there, their difference has at least $d+1$ roots but degree at most $d$; therefore the difference is the zero polynomial. Identification here results from the degree bound, not from finiteness of data alone.
\end{example}

\begin{counterexample}[Interpolation is not unrestricted confirmation]
Given any finite scalar dataset with distinct inputs, a polynomial interpolant exists. Infinitely many smooth functions share those values and differ elsewhere. Exact fit therefore does not establish that the polynomial, rather than another interpolant, is physically realized outside the tested inputs.
\end{counterexample}

\section{Noise, falsification, and scope}
\begin{definition}[Compatibility region]
For observations $\widehat y_x$ and tolerances $\delta_x\geq0$, define
\[
C(D)=\{m\in\mathcal M: d_Y(m(x),\widehat y_x)\leq\delta_x
\text{ for every }x\in T\}.
\]
A model outside $C(D)$ is incompatible with the record relative to the adopted error specification.
\end{definition}

\begin{proposition}[Tolerance monotonicity]
If $\delta'_x\geq\delta_x$ for every tested $x$, then $C(T,\delta)\subseteq C(T,\delta')$.
\end{proposition}
\begin{proof}
Every inequality bounded by $\delta_x$ is also bounded by the larger number $\delta'_x$.
\end{proof}

\begin{interpretation}
Rejection is conditional on the observation model, calibration, and scope. Compatibility is weaker than truth: $C(D)$ can contain many mutually incompatible extrapolations.
\end{interpretation}

\begin{definition}[Scope claim]
A scope claim is an explicit subset $S\subseteq X$ on which a model is asserted to meet a declared predictive criterion. Evidence on $T$ directly addresses the claim only where $T$ probes $S$; extension beyond $T$ invokes further assumptions.
\end{definition}

\begin{assumption}[Regularity-based extension]
Suppose $m$ and the target response $f$ are $L$-Lipschitz on a metric space $S$, and every point of $S$ lies within distance $r$ of some tested point.
\end{assumption}

\begin{proposition}[A bounded extrapolation result]
If $|m(t)-f(t)|\leq\epsilon$ for all tested $t$, then
\[
|m(x)-f(x)|\leq\epsilon+2Lr
\qquad(x\in S).
\]
\end{proposition}
\begin{proof}
Choose $t$ with $d(x,t)\leq r$. The triangle inequality gives
\[
|m(x)-f(x)|\leq |m(x)-m(t)|+|m(t)-f(t)|+|f(t)-f(x)|.
\]
The two Lipschitz bounds contribute at most $Lr$ each, and the tested error contributes at most $\epsilon$.
\end{proof}

\begin{interpretation}
The proposition demonstrates the role of bridge assumptions: data plus regularity and coverage yield a quantitative scope bound. Whether the common Lipschitz bound is adequate is an empirical or independently justified modelling question.
\end{interpretation}

\section{Composition and hidden structure}
\begin{definition}[Componentwise test]
For components $A$ and $B$, a componentwise test records their responses under isolated interfaces. A composite test records the response of $A\otimes B$ after connection through a specified interface.
\end{definition}

\begin{counterexample}[Components do not determine every composite]
Let isolated binary components output unbiased bits. Model I makes the composite outputs independent; Model II makes them perfectly correlated while preserving each isolated marginal. Both agree on every isolated single-component statistic but disagree on the joint event that the outputs coincide. Componentwise adequacy alone does not determine composite behaviour.
\end{counterexample}

\begin{assumption}[Compositional closure]
Predictions for a composite are determined by component models and the declared interface, with no unmodelled shared cause, memory channel, or context dependence across that interface.
\end{assumption}

\begin{empirical}[Interface completeness]
Tests for cross-component correlations, memory, leakage, and contextual dependence provide empirical evidence for or against compositional closure. The categorical availability of a tensor product does not establish interface completeness.
\end{empirical}

\section{Causal and theoretical limits}
\begin{counterexample}[Observation does not fix causal direction]
For correlated Gaussian variables $X,Y$, the same joint distribution can be factorized as $P(X)P(Y\mid X)$ or $P(Y)P(X\mid Y)$. Without intervention, temporal information, or structural restrictions, the observational distribution alone does not choose between the corresponding causal directions.
\end{counterexample}

\begin{assumption}[Causal identification]
A causal conclusion must state its intervention semantics and the assumptions used for identification, such as randomization, absence of specified confounding, an instrumental-variable condition, or a structural causal model.
\end{assumption}

\begin{conjecture}[Hierarchical empirical realization]
A mature compositional physical theory can attach validated uncertainty and scope metadata to components and propagate them through compositions without uncontrolled loss. This is conjectural because dependence, model discrepancy, and interface misspecification can defeat naive propagation.
\end{conjecture}

\section{Dependency ledger and conclusion}
The equivalence and quotient results depend only on set-theoretic definitions. The underdetermination theorem depends on perturbation closure; its smooth corollary uses bump functions. The extrapolation bound depends on explicit Lipschitz and coverage assumptions. Causal and compositional conclusions require additional structural premises whose adequacy is empirical. Accordingly, mathematical admissibility, finite-record compatibility, identifiability within a class, and physical realization are four different statuses.

\paragraph{Bibliographic note.}
The underdetermination of theories by evidence has a long methodological history \cite{Duhem,Glymour}. Statistical identifiability is treated systematically by Lehmann and Casella \cite{LC}, while causal identification and its assumptions are developed by Pearl \cite{Pearl}. The formal results in this paper are elementary and are included to make their dependencies explicit.
