\begin{abstract}
We separate three questions often conflated under Hamiltonian reconstruction: equivalence of regular Lagrangian and Hamiltonian descriptions, recovery of a vector field from ideal trajectories, and statistical identification from finite noisy observations. We prove the regular Legendre correspondence, characterize gauge freedoms visible to trajectory data, and prove non-identifiability for unrestricted Hamiltonians sampled on a finite set. Reconstruction becomes a physical claim only after explicit assumptions concerning state variables, clocks, interventions, and noise.
\end{abstract}

\section{Geometric setting}
Let $Q$ be a smooth finite-dimensional configuration manifold, $TQ$ its tangent bundle, and $T^*Q$ its cotangent bundle. In local coordinates write $(q^i,v^i)$ on $TQ$ and $(q^i,p_i)$ on $T^*Q$.

\begin{definition}[Regular Lagrangian]
A $C^2$ function $L:TQ\to\mathbb R$ is \emph{regular} when its fibre Hessian
\[
W_{ij}(q,v)=\frac{\partial^2L}{\partial v^i\partial v^j}
\]
is nonsingular at every point under consideration.
\end{definition}

\begin{definition}[Legendre map]
The fibre derivative $\mathbb FL:TQ\to T^*Q$ is defined locally by
\[
\mathbb FL(q,v)=(q,p),\qquad p_i=\frac{\partial L}{\partial v^i}(q,v).
\]
If this map is a diffeomorphism onto its image, define
\[
H(q,p)=p_iv^i-L(q,v),\qquad (q,v)=(\mathbb FL)^{-1}(q,p).
\]
\end{definition}

\begin{assumption}[Model domain]
The observed system is represented on a known finite-dimensional manifold $Q$, the reported time parameter corresponds to the model time, and the dynamics are autonomous on the observation interval.
\end{assumption}

\begin{empirical}[Adequacy input]
The model-domain assumption, the choice of coordinates, and the approximation of a laboratory system as closed and autonomous are empirical idealizations. Their adequacy must be tested; it is not implied by the following theorems.
\end{empirical}

\section{The regular Legendre correspondence}
\begin{lemma}
For the Hamiltonian defined above,
\[
\frac{\partial H}{\partial p_i}=v^i,
\qquad
\frac{\partial H}{\partial q^i}=-\frac{\partial L}{\partial q^i},
\]
where $v=v(q,p)$.
\end{lemma}
\begin{proof}
Differentiate $H=p_jv^j-L(q,v)$:
\[
dH=v^jdp_j+p_jdv^j-\frac{\partial L}{\partial q^i}dq^i-
\frac{\partial L}{\partial v^j}dv^j.
\]
By the definition $p_j=\partial L/\partial v^j$, the two terms proportional to $dv^j$ cancel. Comparing coefficients of $dp_i$ and $dq^i$ gives the identities.
\end{proof}

\begin{theorem}[Regular reconstruction]
Suppose $L$ is regular and $\mathbb FL$ is a diffeomorphism onto its image. A curve $q(t)$ satisfies the Euler--Lagrange equations
\[
\frac{d}{dt}\frac{\partial L}{\partial v^i}=\frac{\partial L}{\partial q^i}
\]
if and only if $(q(t),p(t))=\mathbb FL(q(t),\dot q(t))$ satisfies Hamilton's equations
\[
\dot q^i=\frac{\partial H}{\partial p_i},\qquad
\dot p_i=-\frac{\partial H}{\partial q^i}.
\]
\end{theorem}
\begin{proof}
Along the Legendre image, the lemma gives $\partial H/\partial p_i=v^i=\dot q^i$. The definition of momentum gives $\dot p_i=d(\partial L/\partial v^i)/dt$. Thus the Euler--Lagrange equation is equivalent to $\dot p_i=\partial L/\partial q^i=-\partial H/\partial q^i$, again using the lemma. Each step is reversible because the Legendre map is invertible on the stated domain.
\end{proof}

\begin{corollary}[Energy conservation]
If $H$ has no explicit time dependence, every Hamiltonian solution conserves $H$.
\end{corollary}
\begin{proof}
Along a solution,
\[
\frac{dH}{dt}=\frac{\partial H}{\partial q^i}\dot q^i+
\frac{\partial H}{\partial p_i}\dot p_i
=\frac{\partial H}{\partial q^i}\frac{\partial H}{\partial p_i}-
\frac{\partial H}{\partial p_i}\frac{\partial H}{\partial q^i}=0.
\]
\end{proof}

\begin{interpretation}
This theorem reconstructs an equivalent mathematical description from an already specified regular Lagrangian. It does not reconstruct a physically correct Hamiltonian from measurements.
\end{interpretation}

\section{Reconstruction from vector fields}
Let $\omega=dq^i\wedge dp_i$ be the canonical symplectic form. We use the convention $\iota_{X_H}\omega=dH$.

\begin{proposition}[Existence criterion]
Let $X$ be a smooth vector field on a connected symplectic manifold $(M,\omega)$. A global Hamiltonian $H$ with $\iota_X\omega=dH$ exists exactly when the one-form $\iota_X\omega$ is exact. If it exists, it is unique up to an additive constant.
\end{proposition}
\begin{proof}
The first assertion restates the definition of exactness. If $H$ and $K$ generate $X$, then $d(H-K)=0$. A smooth function with zero differential on a connected manifold is constant, proving uniqueness modulo constants.
\end{proof}

\begin{corollary}[Local criterion]
On a contractible open set $U$, a vector field $X$ is Hamiltonian whenever $d(\iota_X\omega)=0$.
\end{corollary}
\begin{proof}
The Poincar\'e lemma makes every closed one-form on $U$ exact. Apply the proposition.
\end{proof}

\begin{counterexample}[Symplectic but not globally Hamiltonian]
On the two-torus with angular coordinates $(q,p)$ and $\omega=dq\wedge dp$, take $X=\partial_p$. Then $\iota_X\omega=-dq$, which is closed but not exact on the torus. Hence local Hamiltonians exist, while no single-valued global Hamiltonian does.
\end{counterexample}

\section{Finite-data non-identifiability}
\begin{definition}[Observed jet data]
For sampled phase points $S=\{z_1,\ldots,z_N\}\subset M$, ideal instantaneous velocity data specify $X_H(z_k)$, equivalently $dH(z_k)=\iota_{X_H(z_k)}\omega$.
\end{definition}

\begin{theorem}[Finite samples do not identify an unrestricted Hamiltonian]
Let $M$ be a smooth symplectic manifold of positive dimension and $S\subset M$ finite. For every smooth Hamiltonian $H$ there are infinitely many smooth Hamiltonians $H_\lambda$, not differing from $H$ merely by constants, whose Hamiltonian vector fields agree with $X_H$ at every point of $S$.
\end{theorem}
\begin{proof}
Choose an open coordinate ball $U$ disjoint from $S$; such a ball exists because a finite set has empty interior in a positive-dimensional manifold. Let $b$ be a nonzero smooth bump function supported in $U$. For $\lambda\in\mathbb R$, set $H_\lambda=H+\lambda b$. Since $b$ vanishes on a neighborhood of every $z_k$, $db(z_k)=0$. Nondegeneracy of $\omega$ then gives $X_{H_\lambda}(z_k)=X_H(z_k)$. For distinct $\lambda$, the differences $(\lambda-\mu)b$ are nonconstant, so the family is inequivalent under additive constants.
\end{proof}

\begin{proposition}[Identification in a finite linear class]
Assume $H_\theta=\sum_{j=1}^m\theta_j\phi_j$ for known smooth basis functions. Exact observations $dH(z_k)$ identify $\theta$ uniquely if and only if the linear map
\[
A:\mathbb R^m\to\bigoplus_{k=1}^NT^*_{z_k}M,
\qquad A\theta=(\sum_j\theta_jd\phi_j(z_k))_k
\]
has trivial kernel.
\end{proposition}
\begin{proof}
Two parameters $\theta$ and $\theta'$ fit the same derivative data exactly when $A(\theta-\theta')=0$. Uniqueness for every attainable datum is therefore equivalent to $\ker A=\{0\}$.
\end{proof}

\begin{example}[Harmonic oscillator]
For $H_{a,b}(q,p)=\tfrac12(ap^2+bq^2)$, Hamilton's equations are $\dot q=ap$ and $\dot p=-bq$. A sample with both $p\neq0$ and $q\neq0$ identifies $a=\dot q/p$ and $b=-\dot p/q$ under exact observation. Samples restricted to $q=0$ contain no information about $b$.
\end{example}

\begin{counterexample}[Clock ambiguity]
If only an unparameterized orbit is observed, multiplying a Hamiltonian vector field by a positive function can preserve the oriented curve while changing its time parametrization. Thus orbit geometry alone need not identify rates. A calibrated clock or equivalent timing assumption is necessary.
\end{counterexample}

\section{From mathematics to inference}
\begin{assumption}[Statistical reconstruction]
A reconstruction procedure specifies a hypothesis class $\mathcal H$, an observation map, a noise model, a sampling design, and a loss or likelihood. Claims of consistency additionally assume that the data-generating process belongs to, or is suitably approximated by, this specification.
\end{assumption}

\begin{empirical}[Validation requirement]
Out-of-sample prediction, residual structure, intervention response, and robustness to state-estimation error are empirical diagnostics. Passing them supports realization only relative to the tested domain and uncertainty budget.
\end{empirical}

\begin{conjecture}[Structured recovery]
Sparse recovery can identify a Hamiltonian in a fixed dictionary when the induced design satisfies an appropriate restricted-injectivity condition and errors are controlled. This statement is conjectural here because no universal sampling or noise assumptions are imposed.
\end{conjecture}

\section{Dependency ledger and conclusion}
The Legendre theorem depends on $C^2$ regularity and invertibility of the fibre derivative. Vector-field reconstruction depends on a specified symplectic form and topology. Data-based identification depends on the hypothesis class and observation operator. Physical realization depends further on empirical adequacy of the state space, clock, closure approximation, and error model. Exact mathematical equivalence and empirical reconstruction are therefore distinct results.

\paragraph{Bibliographic note.}
The geometric mechanics used here is standard; see Abraham and Marsden \cite{AM} and Arnold \cite{Arnold}. Modern data-driven Hamiltonian methods provide useful constrained hypothesis classes, but their empirical success does not remove the identifiability conditions proved above.
